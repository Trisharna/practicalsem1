  void bubblesort(int x[], int n)
{
  int hold,j,pass,k=TRUE;
  for(pass=0; pass<n-1&&k==TRUE; pass++)
  {
    k=FALSE;
    for(j=0; j<n-pass-1; j++) if(x[j]>x[j+1])
       {
	  k=TRUE;
	  hold=x[j];
	  x[j]=x[j+1];
	  x[j+1]=hold;
       }

  }
}
