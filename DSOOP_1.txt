#include<stdio.h>
#include<conio.h>

{

printf("Enter Array Elements:\n");
for(i=0;i<n;i++)
scanf("%d",&a[i]);
printf("Enter element to locate:");
scanf("%d",&val);
found=0;
for(i=0;i<n;i++)
{
if(a[i]==val)
{
printf("Enter found at position%d",i);

}
}
if(found==0)
printf("\nElement not found");

}
